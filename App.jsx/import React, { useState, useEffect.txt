import React, { useState, useEffect, useCallback, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from 'firebase/auth';
import { getFirestore, doc, collection, query, where, addDoc, updateDoc, deleteDoc, onSnapshot, serverTimestamp, getDocs, setDoc } from 'firebase/firestore';
import { Utensils, Users, Lightbulb, X, Trash2, Plus, CornerDownLeft, Swords, Copy, UserPlus, Sparkles, Zap, Shield, Crown, Cake, AlertTriangle, ChevronsDown, Tag } from 'lucide-react';

// --- GLOBAL VARIABLES (Provided by the Canvas Environment) ---
const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';
const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : null;
const initialAuthToken = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;
// -----------------------------------------------------------

// Gemini API Configuration
const apiKey = ""; // API Key is provided by the environment
const GEMINI_MODEL = "gemini-2.5-flash-preview-05-20";
const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${GEMINI_MODEL}:generateContent?key=${apiKey}`;

// Global Allergy List
const ALLERGIES = ['Peanuts', 'Tree Nuts', 'Milk', 'Eggs', 'Wheat/Gluten', 'Soy', 'Fish', 'Shellfish', 'Other'];

// Utility to get a simple display name from the UID
const getDisplayName = (uid) => uid ? `User-${uid.substring(0, 4)}` : 'You';
const isKidProfile = (profileId) => profileId !== 'self';

// State Management for UI Messages (Non-firebase alerts)
const useMessage = () => {
  const [message, setMessage] = useState(null);
  const showMessage = useCallback((text, type = 'info', duration = 3000) => {
    setMessage({ text, type });
    setTimeout(() => setMessage(null), duration);
  }, []);
  return [message, showMessage];
};

// Main Application Component
const App = () => {
  const [db, setDb] = useState(null);
  const [auth, setAuth] = useState(null);
  const [userId, setUserId] = useState(null);
  const [isAuthReady, setIsAuthReady] = useState(false);
  const [activeTab, setActiveTab] = useState('menu');
  const [userMenus, setUserMenus] = useState([]);
  const [familyProfiles, setFamilyProfiles] = useState([]); 
  const [groups, setGroups] = useState([]);
  const [message, showMessage] = useMessage();
  const [currentSuggestion, setCurrentSuggestion] = useState(null);
  const [userSettings, setUserSettings] = useState({ role: 'Single User' }); // NEW STATE
  
  // LLM Feature States
  const [isLlmLoading, setIsLlmLoading] = useState(false);
  const [expandedOrder, setExpandedOrder] = useState(null);
  const [compromiseSuggestion, setCompromiseSuggestion] = useState(null);
  const [inviteMessage, setInviteMessage] = useState(null); 
  const [suggestionJustification, setSuggestionJustification] = useState(null);
  
  // Memoized path constants
  const getMenuCollectionPath = useCallback((uid) => `artifacts/${appId}/users/${uid}/user_menus`, []);
  const getFamilyProfilesCollectionPath = useCallback((uid) => `artifacts/${appId}/users/${uid}/family_profiles`, []);
  const getGroupCollectionPath = useMemo(() => `artifacts/${appId}/public/data/groups`, []);
  // NEW PATH
  const getUserSettingsDocPath = useCallback((uid) => doc(db, `artifacts/${appId}/users/${uid}/user_settings`, 'main_settings'), [db]);

  // 1. Firebase Initialization and Authentication
  useEffect(() => {
    if (firebaseConfig) {
      try {
        const app = initializeApp(firebaseConfig);
        const firestoreDb = getFirestore(app);
        const firebaseAuth = getAuth(app);
        setDb(firestoreDb);
        setAuth(firebaseAuth);

        const unsubscribe = onAuthStateChanged(firebaseAuth, async (user) => {
          if (user) {
            setUserId(user.uid);
          } else {
            try {
              if (initialAuthToken) {
                const credential = await signInWithCustomToken(firebaseAuth, initialAuthToken);
                setUserId(credential.user.uid);
              } else {
                const credential = await signInAnonymously(firebaseAuth);
                setUserId(credential.user.uid);
              }
            } catch (error) {
              console.error("Authentication Error:", error);
              const credential = await signInAnonymously(firebaseAuth);
              setUserId(credential.user.uid);
            }
          }
          setIsAuthReady(true);
        });
        return () => unsubscribe();
      } catch (e) {
        console.error("Firebase Initialization Error:", e);
        showMessage("Failed to initialize app.", "error");
      }
    }
  }, [firebaseConfig, initialAuthToken, showMessage]);

  // 1.5 Fetch User Settings (NEW)
  useEffect(() => {
    if (db && userId) {
        const docRef = doc(db, `artifacts/${appId}/users/${userId}/user_settings`, 'main_settings');
        const unsubscribe = onSnapshot(docRef, (docSnap) => {
            if (docSnap.exists()) {
                setUserSettings(docSnap.data());
            } else {
                // Initialize settings if they don't exist
                const initialSettings = { role: 'Single User' };
                setUserSettings(initialSettings);
                // Attempt to create the document (non-blocking)
                setDoc(docRef, initialSettings).catch(e => console.error("Error initializing user settings:", e));
            }
        }, (error) => {
            console.error("Error fetching user settings:", error);
        });
        return () => unsubscribe();
    }
  }, [db, userId, showMessage]);

  // Function to update the user's role
  const updateRole = async (newRole) => {
      if (!db || !userId) return showMessage("App is not ready.", "error");
      try {
          const docRef = doc(db, `artifacts/${appId}/users/${userId}/user_settings`, 'main_settings');
          await updateDoc(docRef, { role: newRole });
          showMessage(`Primary role updated to ${newRole}!`, "success");
      } catch (e) {
          console.error("Error updating role:", e);
          showMessage("Could not update role.", "error");
      }
  };


  // 2. Fetch User's Personal Menus (Private Data) - UNCHANGED
  useEffect(() => {
    if (db && userId) {
      const q = collection(db, getMenuCollectionPath(userId));
      const unsubscribe = onSnapshot(q, (snapshot) => {
        const items = snapshot.docs.map(doc => ({
          id: doc.id,
          ...doc.data(),
        }));
        setUserMenus(items);
      }, (error) => {
        console.error("Error fetching user menus:", error);
        showMessage("Failed to load your personal menu.", "error");
      });
      return () => unsubscribe();
    }
  }, [db, userId, getMenuCollectionPath, showMessage]);
  
  // 3. Fetch Family Profiles (UPDATED)
  useEffect(() => {
    if (db && userId) {
      const q = collection(db, getFamilyProfilesCollectionPath(userId));
      const unsubscribe = onSnapshot(q, (snapshot) => {
        const profiles = snapshot.docs.map(doc => ({
          id: doc.id,
          ...doc.data(),
        }));
        setFamilyProfiles(profiles); // Use setFamilyProfiles
      }, (error) => {
        console.error("Error fetching family profiles:", error);
      });
      return () => unsubscribe();
    }
  }, [db, userId, getFamilyProfilesCollectionPath]);


  // 4. Fetch Groups (Public Data) - UNCHANGED
  useEffect(() => {
    if (db && userId && isAuthReady) {
      const q = query(collection(db, getGroupCollectionPath), where('memberUids', 'array-contains', userId));
      const unsubscribe = onSnapshot(q, (snapshot) => {
        const groupList = snapshot.docs.map(doc => ({
          id: doc.id,
          ...doc.data(),
        }));
        setGroups(groupList);
      }, (error) => {
        console.error("Error fetching groups:", error);
        showMessage("Failed to load your groups.", "error");
      });
      return () => unsubscribe();
    }
  }, [db, userId, isAuthReady, getGroupCollectionPath, showMessage]);
  
  // Utility function to get the profile name (self or family member's name)
  const getProfileName = (profileId) => {
    if (profileId === 'self') return getDisplayName(userId) || 'You';
    const profile = familyProfiles.find(p => p.id === profileId);
    return profile ? profile.name : 'Unknown';
  };
  
  const getProfile = (profileId) => {
    if (profileId === 'self') return { name: getDisplayName(userId), type: 'Adult', allergies: [], birthday: '' };
    return familyProfiles.find(p => p.id === profileId);
  };

  // --- LLM API Call Utility (unchanged) ---
  const fetchGeminiResponse = async (userQuery, systemPrompt) => {
    setIsLlmLoading(true);
    let attempts = 0;
    const maxAttempts = 3;

    while (attempts < maxAttempts) {
      try {
        const payload = {
          contents: [{ parts: [{ text: userQuery }] }],
          systemInstruction: { parts: [{ text: systemPrompt }] },
        };

        const response = await fetch(apiUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });

        if (!response.ok) {
          throw new Error(`HTTP error! status: ${response.status}`);
        }

        const result = await response.json();
        const text = result.candidates?.[0]?.content?.parts?.[0]?.text || 'Could not generate a response.';
        setIsLlmLoading(false);
        return text;

      } catch (error) {
        attempts++;
        if (attempts >= maxAttempts) {
          console.error("Gemini API call failed after multiple retries:", error);
          setIsLlmLoading(false);
          showMessage("LLM service is temporarily unavailable.", "error");
          return 'Error: Failed to get response from AI.';
        }
        const delay = Math.pow(2, attempts) * 1000;
        await new Promise(resolve => setTimeout(resolve, delay));
      }
    }
  };

  const generateOrderDetails = async (restaurant, order) => {
    setExpandedOrder(null);
    setActiveTab('menu');
    const systemPrompt = `You are a highly creative and detailed culinary analyst. Expand the user's simple go-to order (Restaurant: ${restaurant}, Order: ${order}) into a professional, mouth-watering description, including recommended side dishes, suggested drink pairings, and a short (one-line) fun fact or history about the dish or cuisine type. Use bold markdown for key details.`;
    const userQuery = `Expand this order: ${order} from ${restaurant}`;
    
    const details = await fetchGeminiResponse(userQuery, systemPrompt);
    setExpandedOrder({ restaurant, order, details });
  };
  
  const generateGroupCompromise = async (group) => {
    setCompromiseSuggestion(null);
    
    // Flatten orders for the LLM
    const uniqueRestaurants = [...new Set(group.restaurantOrders.map(r => r.restaurantName))];
    const allOrders = group.restaurantOrders.flatMap(r => 
        r.orders.map(o => `${r.restaurantName} (${getDisplayName(o.userId)}): ${o.order}`)
    ).join('\n');
    
    if (uniqueRestaurants.length < 2) {
        showMessage("Need at least two different restaurant types to suggest a compromise.", "warning");
        return;
    }

    const systemPrompt = `You are a professional diplomatic group mediator and a gourmet food expert. Based on the following diverse list of preferred restaurants/orders from a group, suggest a single, ideal compromise restaurant type or cuisine that incorporates elements from the different tastes. Explain your choice in one short, engaging paragraph and suggest a specific dish everyone should try at the compromise location. Format your output in markdown.`;
    const userQuery = `Group orders:\n${allOrders}`;
    
    const suggestion = await fetchGeminiResponse(userQuery, systemPrompt);
    setCompromiseSuggestion({ groupName: group.name, suggestion });
  };
  
  const generateGroupInviteMessage = async (group) => {
    setInviteMessage(null);
    
    // Flatten orders for the LLM
    const uniqueRestaurants = [...new Set(group.restaurantOrders.map(r => r.restaurantName))];
    const allOrders = group.restaurantOrders.flatMap(r => 
        r.orders.map(o => `${r.restaurantName} (${getDisplayName(o.userId)}): ${o.order}`)
    ).join('; ');
    
    if (uniqueRestaurants.length === 0) {
        showMessage("No orders saved in this group yet. Add some first!", "warning");
        return;
    }

    const systemPrompt = `You are a fun, casual, and slightly sarcastic friend sending a group chat message to finalize dinner plans based on saved go-to orders. Generate a short, one-paragraph message (under 3 sentences) that confirms the restaurant choices and asks everyone to finalize their order. Include a fun emoji. The style should be suitable for a text message/WhatsApp group.`;
    const userQuery = `The group has these current saved orders: ${allOrders}. The primary restaurants suggested are: ${uniqueRestaurants.join(', ')}.`;
    
    const message = await fetchGeminiResponse(userQuery, systemPrompt);
    setInviteMessage({ groupName: group.name, message });
    showMessage("Invite message generated!", "success");
  };

  const generateSuggestionJustification = async (restaurant, order) => {
    const systemPrompt = "You are a highly enthusiastic, motivational food guru/decision-maker. Your job is to convince the user why they should eat the suggested meal right now based on current life/mood factors. Generate a catchy title (using a fun emoji) and a short, punchy paragraph (under 3 sentences) justifying the meal based on reasons like 'you deserve a treat,' 'it's fast and easy,' 'perfect for this weather,' or 'a flavor adventure.' Use bold markdown for key details.";
    const userQuery = `Justify why I should eat: ${order} from ${restaurant}.`;
    
    const justification = await fetchGeminiResponse(userQuery, systemPrompt);
    return justification;
  };

  // --- Utility Functions (unchanged) ---

  const copyText = (text, successMessage) => {
    if (!text) return;
    const el = document.createElement('textarea');
    el.value = text;
    document.body.appendChild(el);
    el.select();
    try {
      document.execCommand('copy');
      showMessage(successMessage, "success");
    } catch (err) {
      console.error('Could not copy text: ', err);
      showMessage("Failed to copy. Please select and copy manually.", "error");
    }
    document.body.removeChild(el);
  };
  
  const copyUserId = () => {
    if (!userId) return;
    copyText(userId, "User ID copied to clipboard!");
  };

  // --- CRUD Operations for Family Profiles (UPDATED) ---

  const addFamilyProfile = async ({ name, type, birthday, allergies }) => {
    if (!db || !userId) return showMessage("App is not ready.", "error");
    const trimmedName = name.trim();
    if (!trimmedName) return showMessage("Profile name cannot be empty.", "warning");
    
    try {
      await addDoc(collection(db, getFamilyProfilesCollectionPath(userId)), {
        name: trimmedName,
        type: type, // 'Kid' or 'Adult'
        birthday: birthday, // Month/Day string
        allergies: allergies, // Array of strings
        createdAt: serverTimestamp(),
      });
      showMessage(`${trimmedName} profile (${type}) added!`, "success");
    } catch (e) {
      console.error("Error adding family profile: ", e);
      showMessage("Could not add profile.", "error");
    }
  };

  const deleteFamilyProfile = async (id) => {
    if (!db || !userId) return showMessage("App is not ready.", "error");
    try {
      // Find and delete all menu items belonging to this profile first
      const q = query(collection(db, getMenuCollectionPath(userId)), where('profileId', '==', id));
      const snapshot = await getDocs(q); 
      snapshot.forEach(async (doc) => {
        await deleteDoc(doc.ref);
      });
      
      // Then delete the profile itself
      await deleteDoc(doc(db, getFamilyProfilesCollectionPath(userId), id));
      showMessage("Profile and associated orders removed.", "success");
    } catch (e) {
      console.error("Error removing family profile: ", e);
      showMessage("Could not remove profile.", "error");
    }
  };

  // --- CRUD Operations for Personal Menu (UNCHANGED logic, UPDATED usage) ---

  const addMenuItem = async ({ name, order, profileId }) => {
    if (!db || !userId) return showMessage("App is not ready.", "error");
    try {
      await addDoc(collection(db, getMenuCollectionPath(userId)), {
        name: name.trim(),
        order: order.trim(),
        profileId: profileId || 'self', // Default to 'self'
        createdAt: serverTimestamp(),
      });
      showMessage("Order saved!", "success");
    } catch (e) {
      console.error("Error adding document: ", e);
      showMessage("Could not save order.", "error");
    }
  };

  const deleteMenuItem = async (id) => {
    if (!db || !userId) return showMessage("App is not ready.", "error");
    try {
      await deleteDoc(doc(db, getMenuCollectionPath(userId), id));
      showMessage("Order removed.", "success");
    } catch (e) {
      console.error("Error removing document: ", e);
      showMessage("Could not remove order.", "error");
    }
  };

  // --- Group Management Functions (UNCHANGED) ---

  const createGroup = async (groupName) => {
    if (!db || !userId) return showMessage("App is not ready.", "error");
    const newGroupName = groupName.trim();
    if (!newGroupName) return showMessage("Group name cannot be empty.", "warning");

    try {
      await addDoc(collection(db, getGroupCollectionPath), {
        name: newGroupName,
        memberUids: [userId],
        members: [{ uid: userId, name: getDisplayName(userId) }],
        restaurantOrders: [], 
        createdAt: serverTimestamp(),
      });
      showMessage(`Group "${newGroupName}" created!`, "success");
    } catch (e) {
      console.error("Error creating group: ", e);
      showMessage("Could not create group.", "error");
    }
  };

  const addOrderToGroup = async (groupId, restaurantName, orderDescription) => {
    if (!db || !userId) return showMessage("App is not ready.", "error");
    const restName = restaurantName.trim();
    const orderDesc = orderDescription.trim();
    if (!restName || !orderDesc) return showMessage("Restaurant and order cannot be empty.", "warning");

    try {
      const groupRef = doc(db, getGroupCollectionPath, groupId);
      const group = groups.find(g => g.id === groupId);

      if (!group) return showMessage("Group not found.", "error");

      let updatedRestaurantOrders = [...(group.restaurantOrders || [])];
      const existingRestaurantIndex = updatedRestaurantOrders.findIndex(r => r.restaurantName.toLowerCase() === restName.toLowerCase());

      if (existingRestaurantIndex > -1) {
        let orders = updatedRestaurantOrders[existingRestaurantIndex].orders || [];
        const existingOrderIndex = orders.findIndex(o => o.userId === userId);

        if (existingOrderIndex > -1) {
          orders[existingOrderIndex] = { userId, order: orderDesc };
        } else {
          orders.push({ userId, order: orderDesc });
        }
        updatedRestaurantOrders[existingRestaurantIndex].orders = orders;
      } else {
        updatedRestaurantOrders.push({
          restaurantName: restName,
          orders: [{ userId, order: orderDesc }],
          votes: [],
        });
      }

      await updateDoc(groupRef, { restaurantOrders: updatedRestaurantOrders });
      showMessage(`Your order for ${restName} saved in the group!`, "success");
    } catch (e) {
      console.error("Error adding order to group: ", e);
      showMessage("Could not save group order.", "error");
    }
  };

  const inviteUserToGroup = async (groupId, targetUserId) => {
    if (!db || !userId) return showMessage("App is not ready.", "error");
    const idToInvite = targetUserId.trim();

    if (!idToInvite) return showMessage("Please enter a User ID.", "warning");
    if (idToInvite === userId) return showMessage("That's your ID!", "warning");
    
    try {
        const groupRef = doc(db, getGroupCollectionPath, groupId);
        const group = groups.find(g => g.id === groupId);

        if (!group) return showMessage("Group not found.", "error");
        if (group.memberUids.includes(idToInvite)) return showMessage("User is already a member.", "warning");
        
        const newMember = { uid: idToInvite, name: getDisplayName(idToInvite) };

        await updateDoc(groupRef, {
            memberUids: [...group.memberUids, idToInvite],
            members: [...(group.members || []), newMember]
        });
        showMessage(`User ${getDisplayName(idToInvite)} invited successfully!`, "success");
    } catch (e) {
        console.error("Error inviting user:", e);
        showMessage("Could not invite user.", "error");
    }
  };

  const handleVote = async (groupId, restaurantName) => {
    if (!db || !userId) return showMessage("App is not ready.", "error");
    
    try {
      const groupRef = doc(db, getGroupCollectionPath, groupId);
      const group = groups.find(g => g.id === groupId);
      if (!group) return showMessage("Group not found.", "error");

      let updatedRestaurantOrders = [...(group.restaurantOrders || [])];
      const restIndex = updatedRestaurantOrders.findIndex(r => r.restaurantName === restaurantName);

      if (restIndex > -1) {
        let votes = updatedRestaurantOrders[restIndex].votes || [];
        const userHasVoted = votes.includes(userId);

        if (userHasVoted) {
          updatedRestaurantOrders[restIndex].votes = votes.filter(uid => uid !== userId);
          showMessage(`Unvoted for ${restaurantName}.`, "info");
        } else {
          updatedRestaurantOrders.forEach((r, index) => {
            if (index !== restIndex) {
              r.votes = (r.votes || []).filter(uid => uid !== userId);
            }
          });
          updatedRestaurantOrders[restIndex].votes = [...votes, userId];
          showMessage(`Voted for ${restaurantName}!`, "success");
        }

        await updateDoc(groupRef, { restaurantOrders: updatedRestaurantOrders });

      } else {
        showMessage("Restaurant not found for voting.", "error");
      }
    } catch (e) {
      console.error("Error handling vote: ", e);
      showMessage("Could not process vote.", "error");
    }
  };

  const suggestDinner = async () => {
    if (userMenus.length === 0) {
      setCurrentSuggestion({ restaurant: 'No Saved Menus', order: 'Add some favorite orders first!' });
      setSuggestionJustification(null);
      return;
    }
    
    setSuggestionJustification(null); 
    setCurrentSuggestion(null);
    setIsLlmLoading(true);

    await new Promise(resolve => setTimeout(resolve, 800)); 
    
    // Filter to only include 'self' profile for suggestions
    const selfMenus = userMenus.filter(m => m.profileId === 'self');

    if (selfMenus.length === 0) {
      setIsLlmLoading(false);
      setCurrentSuggestion({ restaurant: 'No *Your* Menus', order: 'Add some go-to orders for yourself!' });
      return;
    }
    
    const randomIndex = Math.floor(Math.random() * selfMenus.length);
    const suggestion = selfMenus[randomIndex];

    setCurrentSuggestion({
      restaurant: suggestion.name,
      order: suggestion.order,
      isNew: true
    });
    
    const justification = await generateSuggestionJustification(suggestion.name, suggestion.order);
    
    setSuggestionJustification(justification);
    setIsLlmLoading(false);
    showMessage("The Master Menu has spoken!", "info");
    setActiveTab('suggest');
  };

  // --- Sub-Components ---
  // ... (LoadingState, MessageBar, LlmLoader are visually unchanged but included)

  const LoadingState = ({ message = "Connecting to the culinary database..." }) => (
    <div className="flex flex-col items-center justify-center p-8 text-center text-gray-500">
      <Utensils className="w-10 h-10 mb-4 animate-spin text-cyan-400" />
      <p className="font-semibold text-lg">{message}</p>
      <p className="text-sm">User ID: {userId ? userId : 'N/A'}</p>
    </div>
  );

  const MessageBar = () => {
    if (!message) return null;
    const baseClasses = "fixed top-4 left-1/2 -translate-x-1/2 p-3 text-sm font-medium rounded-lg shadow-xl z-50 transition-all duration-300";
    const colorClasses = {
      // UPDATED: Orange -> Cyan/Blue for accents
      info: "bg-cyan-600 text-white", 
      success: "bg-emerald-500 text-white",
      error: "bg-red-500 text-white",
      warning: "bg-amber-500 text-white",
    }[message.type] || "bg-gray-700 text-white";

    return (
      <div className={`${baseClasses} ${colorClasses}`}>
        {message.text}
      </div>
    );
  };

  const LlmLoader = () => (
    // UPDATED: Yellow -> Blue for LLM/Info
    <div className="flex items-center justify-center p-4 bg-blue-50 rounded-xl border border-blue-400 shadow-md">
      <Zap className="w-5 h-5 mr-3 text-blue-700 animate-pulse" />
      <p className="text-sm font-semibold text-blue-800">Gemini is deciding for you...</p> {/* FIX: Removed duplicate className */}
    </div>
  );
  
  // Component: Role Selector
  const RoleSelector = () => {
    const roles = [
        { key: 'Single User', label: 'Single User (Personal Menu)' },
        { key: 'Group Manager', label: 'Group Manager (Roommates/Friends)' },
        { key: 'Family Manager & Caregiver', label: 'Family Manager & Caregiver (Profiles/Allergies)' },
    ];

    const currentRole = userSettings.role || 'Single User';

    return (
        <div className="p-5 bg-gray-100 rounded-2xl shadow-inner border border-gray-300">
            <h4 className="text-lg font-bold text-gray-800 mb-3 flex items-center border-b pb-2">
                <Tag className="w-5 h-5 mr-2 text-cyan-600" /> Primary Role Selector
            </h4>
            <p className='text-sm text-gray-600 mb-3'>
                Choose the role that best defines how you use the app. This updates the main header title.
            </p>
            <select
                value={currentRole}
                onChange={(e) => updateRole(e.target.value)}
                className="w-full p-3 border border-gray-300 rounded-xl focus:ring-cyan-500 shadow-sm bg-white font-medium text-lg text-cyan-700"
            >
                {roles.map(role => (
                    <option key={role.key} value={role.key}>
                        {role.label}
                    </option>
                ))}
            </select>
        </div>
    );
  };

  // --- Family Manager Component (NEW/UPDATED) ---

  const AllergySummary = () => {

        {/* List of Groups */}
        <h3 className="text-2xl font-bold text-gray-800 border-b pb-2">Your Groups ({groups.length})</h3>
        <div className="space-y-4">
          {groups.length === 0 ? (
            <div className="p-8 text-center bg-gray-100 rounded-2xl text-gray-700 border-2 border-dashed border-gray-300 shadow-inner"> {/* UPDATED: Stone -> Gray */}
              You are not a member of any groups. Create one above!
            </div>
          ) : (
            groups
              .sort((a, b) => a.name.localeCompare(b.name))
              .map((group) => (
                <div
                  key={group.id}
                  // UPDATED: Orange -> Cyan
                  className="p-5 bg-white border border-gray-100 rounded-2xl shadow-lg flex justify-between items-center cursor-pointer hover:bg-cyan-50 transition duration-200 transform hover:scale-[1.01]"
                  onClick={() => setActiveGroup(group)}
                >
                  <div className="flex-1 min-w-0">
                    <p className="text-xl font-semibold text-gray-800 truncate">{group.name}</p>
                    <p className="text-sm text-gray-500 mt-1">Members: {group.members ? group.members.length : 1}</p>
                  </div>
                  <button
                    className="p-2 text-cyan-600 hover:text-cyan-800 rounded-full transition duration-150"
                  >
                    <CornerDownLeft className="w-5 h-5 rotate-180" />
                  </button>
                </div>
              ))
          )}
        </div>
      </div>
    );
  };

  const SuggestDinner = () => (
    // UPDATED: Orange -> Cyan
    <div className="flex flex-col items-center justify-center p-8 bg-cyan-100 rounded-2xl shadow-2xl border-4 border-cyan-400">
      <h3 className="text-3xl font-extrabold text-cyan-800 mb-6 flex items-center">
        <Lightbulb className="w-8 h-8 mr-3 text-cyan-600 fill-cyan-300" /> {/* UPDATED: Orange -> Cyan */}
      </h3>

      <button
        onClick={suggestDinner}
        // UPDATED: Orange -> Cyan
        className="flex items-center justify-center bg-cyan-800 text-white text-xl p-4 rounded-xl font-bold shadow-2xl hover:bg-cyan-900 transition duration-200 transform hover:scale-[1.05]"
        disabled={userMenus.filter(m => m.profileId === 'self').length === 0 || isLlmLoading}
      >
        <Swords className="w-6 h-6 mr-3" />
        {isLlmLoading ? "Consulting the Culinary Oracle..." : (userMenus.filter(m => m.profileId === 'self').length === 0 ? "Add YOUR Menus to Decide" : "Suggest Dinner Now!")}
      </button>

      <div className="mt-8 w-full">
        {isLlmLoading && <LlmLoader message="The Master Menu is deciding for you..." />}

        {currentSuggestion && !isLlmLoading && (
          <div className="w-full text-center space-y-6">
            
            <div className="p-8 bg-white rounded-xl shadow-xl border-t-4 border-cyan-500 animate-fadeIn"> {/* UPDATED: Orange -> Cyan */}
              <p className="text-md text-gray-500 mb-2 font-medium">The prophecy has been cast:</p>
              <p className="text-5xl font-extrabold text-cyan-700 animate-bounceOnce">{currentSuggestion.restaurant}</p> {/* UPDATED: Orange -> Cyan */}
              <p className="text-2xl text-gray-700 italic mt-3">({currentSuggestion.order})</p>
            </div>

            {suggestionJustification ? (
              <div className="p-6 bg-blue-50 rounded-xl shadow-lg border-l-4 border-blue-500 text-left"> {/* UPDATED: Yellow -> Blue */}
                <h4 className="text-xl font-bold text-blue-800 flex items-center mb-2"> {/* UPDATED: Yellow -> Blue */}
                    <Sparkles className="w-5 h-5 mr-2" />
                    Why This Pick is Perfect
                </h4>
                <div className="prose max-w-none text-gray-700" dangerouslySetInnerHTML={{ __html: suggestionJustification.replace(/\n/g, '<br/>') }}></div>
                <p className="text-xs text-gray-500 italic pt-3 border-t border-blue-200 mt-3">Decision fueled by Gemini AI.</p> {/* UPDATED: Yellow -> Blue */}
              </div>
            ) : (
                <p className="text-gray-500 italic">No justification found.</p>
            )}
          </div>
        )}

        {!currentSuggestion && !isLlmLoading && (
             <p className="mt-5 text-lg text-cyan-700 font-medium">Click the button to let fate decide your next delicious meal.</p>
        )}
      </div>
      
      {userMenus.length > 0 && (
        <p className="mt-5 text-sm text-cyan-700 font-medium">Drawing from {userMenus.length} saved go-to orders.</p>
      )}
    </div>
  );


  // --- Main Render ---

  if (!isAuthReady) {
    return <LoadingState />;
  }

  return (
    <div className="min-h-screen bg-gray-50 font-sans"> {/* UPDATED: Stone -> Gray */}
      <MessageBar />

      <header className="bg-gray-800 text-white p-6 shadow-2xl"> {/* UPDATED: Stone -> Gray */}
        <h1 className="text-4xl font-extrabold flex items-center">
          <Utensils className="w-9 h-9 mr-3 text-cyan-300" /> {/* UPDATED: Amber -> Cyan */}
          The Master Menu
        </h1>
        <p className="text-sm opacity-80 mt-1 font-light">
            Go To Order At All Your Food Spots
        </p>
        {userSettings.role && (
            <span className='mt-2 inline-block px-3 py-1 bg-teal-500 text-white text-xs font-bold rounded-full shadow-md'> {/* UPDATED: Amber -> Teal */}
                {userSettings.role}
            </span>
        )}
        <div className="text-xs mt-3 flex items-center">
          <span className="mr-2 opacity-80 font-mono">Authenticated User ID: {userId}</span>
          <button onClick={copyUserId} className="text-cyan-200 hover:text-white p-1 rounded transition duration-150 transform hover:scale-110" title="Click to copy your User ID"> {/* UPDATED: Amber -> Cyan */}
            <Copy className="w-4 h-4" />
          </button>
        </div>
      </header>

      {/* Navigation Tabs */}
      <nav className="bg-white shadow-lg sticky top-0 z-10">
        <div className="flex justify-around border-b border-gray-200 max-w-4xl mx-auto">
          <TabButton icon={Utensils} label="My Master Menu" tabId="menu" activeTab={activeTab} setActiveTab={setActiveTab} />
          <TabButton icon={Users} label="Groups" tabId="groups" activeTab={activeTab} setActiveTab={setActiveTab} />
          <TabButton icon={Lightbulb} label="Pick For Me" tabId="suggest" activeTab={activeTab} setActiveTab={setActiveTab} />
        </div>
      </nav>

      <main className="p-4 max-w-4xl mx-auto">
        {activeTab === 'menu' && (
          <div className="space-y-6">
            <h2 className="text-3xl font-bold text-gray-800 border-b pb-2 mt-4">My Master Menu</h2>
            <RoleSelector />
            <AddFamilyProfileForm />
            <AllergySummary />
            <AddMenuItemForm />
            <MenuList />
          </div>
        )}

        {activeTab === 'groups' && (
          <div>
            <h2 className="text-3xl font-bold text-gray-800 border-b pb-2 mt-4">Group Orders Tracker</h2>
            <GroupList />
          </div>
        )}

        {activeTab === 'suggest' && (
          <div>
            <h2 className="text-3xl font-bold text-gray-800 border-b pb-2 mt-4">Pick For Me</h2>
            <SuggestDinner />
          </div>
        )}
      </main>
    </div>
  );
};

// Simple reusable component for the tab navigation
const TabButton = ({ icon: Icon, label, tabId, activeTab, setActiveTab }) => {
  const isActive = activeTab === tabId;
  return (
    <button
      onClick={() => setActiveTab(tabId)}
      className={`flex-1 flex flex-col items-center p-4 text-md font-semibold transition duration-200 ${
        isActive
          // UPDATED: Orange -> Cyan
          ? 'text-cyan-700 border-b-4 border-cyan-700 bg-cyan-50 shadow-inner'
          : 'text-gray-500 hover:text-cyan-600 hover:bg-gray-50' // UPDATED: Orange -> Cyan
      }`}
    >
      <Icon className="w-5 h-5 mb-1" />
      {label}
    </button>
  );
};

export default App;
